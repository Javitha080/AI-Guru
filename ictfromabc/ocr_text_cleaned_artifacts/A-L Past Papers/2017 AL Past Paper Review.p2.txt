def calculate_bill(household_no, previous_reading, present_reading):
    # Assumption: present_reading >= previous_reading
    units_consumed = present_reading - previous_reading

    if units_consumed <= 64:
        amount = units_consumed * 5.0
    else:
        amount = (64 * 5.0) + ((units_consumed - 64) * 10.0)

    print(f"Household No: {household_no}")
    print(f"Units Consumed: {units_consumed}")
    print(f"Total Amount: Rs.{amount:.2f}")

    return units_consumed, amount
