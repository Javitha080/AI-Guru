           global datakeys

        def printdata():
           global datakeys, datasummary
           f2 = open('output.txt', 'w')
           for key in datakeys:
              f2.write('{}-{}\n'.format(key,datasummary[key]))
           f2.close()

        readdata()
        processdata()
        printdata()
